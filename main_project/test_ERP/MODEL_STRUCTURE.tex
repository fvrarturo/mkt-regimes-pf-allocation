\section{Introduction}

This document presents the complete structure of a regime-weighted economic forecasting model designed for active asset allocation. The model consists of three integrated modules that work together to generate economic forecasts and translate them into portfolio allocation decisions.

\section{Model Architecture}

The model is structured into three main modules:

\begin{enumerate}
    \item \textbf{Module 1: Regime Transition Forecasting} - Forecasts future market regimes using Hidden Markov Model (HMM) transition probabilities
    \item \textbf{Module 2: Economic Variable Forecasting} - Generates forecasts for key economic variables using regime information and macro variable weighting
    \item \textbf{Module 3: Asset Allocation Engine} - Translates forecasts into actionable portfolio allocation decisions
\end{enumerate}

\section{Module 1: Regime Transition Forecasting}

\subsection{Component 1.1: Current Regime Identification}

\textbf{Input:}
\begin{itemize}
    \item Current economic data (GDP, inflation, unemployment, etc.)
    \item HMM model parameters
\end{itemize}

\textbf{Process:}
The current regime probabilities are computed using the HMM forward algorithm:
\begin{equation}
    P(\text{Regime}_j | \mathcal{I}_t) = \frac{\alpha_t(j)}{\sum_{k=0}^{3} \alpha_t(k)}
\end{equation}

where $\mathcal{I}_t$ represents the information set at time $t$, and $\alpha_t(j)$ is the forward probability for regime $j$ at time $t$.

\textbf{Output:}
\begin{equation}
    \mathbf{p}_t = [P(R_0), P(R_1), P(R_2), P(R_3)]
\end{equation}

where $\sum_{j=0}^{3} P(R_j) = 1$.

\subsection{Component 1.2: Transition Probability Matrix}

\textbf{Input:}
\begin{itemize}
    \item Historical regime sequence from HMM
    \item Estimated transition matrix: $P(\text{Regime}_{t+1} | \text{Regime}_t)$
\end{itemize}

\textbf{Process:}
The transition matrix $\mathbf{A}$ is extracted from the HMM model:
\begin{equation}
    \mathbf{A} = \begin{bmatrix}
        a_{00} & a_{01} & a_{02} & a_{03} \\
        a_{10} & a_{11} & a_{12} & a_{13} \\
        a_{20} & a_{21} & a_{22} & a_{23} \\
        a_{30} & a_{31} & a_{32} & a_{33}
    \end{bmatrix}
\end{equation}

where $a_{ij} = P(\text{Regime}_j \text{ at } t+1 | \text{Regime}_i \text{ at } t)$ and $\sum_{j=0}^{3} a_{ij} = 1$ for all $i$.

\textbf{Output:}
\begin{itemize}
    \item 1-step transition matrix $\mathbf{A}$
    \item Multi-step transition matrices $\mathbf{A}^h$ for horizons $h \in \{1, 3, 6, 12\}$ months
\end{itemize}

\subsection{Component 1.3: Future Regime Probability Forecasting}

\textbf{Input:}
\begin{itemize}
    \item Current regime probabilities $\mathbf{p}_t$ (from Component 1.1)
    \item Transition matrix $\mathbf{A}$ (from Component 1.2)
    \item Forecast horizon $h \in \{1, 3, 6, 12\}$ months
\end{itemize}

\textbf{Process:}
Future regime probabilities are computed as:
\begin{equation}
    \mathbf{p}_{t+h} = \mathbf{p}_t \times \mathbf{A}^h
\end{equation}

For each horizon $h$, this yields:
\begin{equation}
    P(\text{Regime}_j \text{ at } t+h) = \sum_{i=0}^{3} P(\text{Regime}_i \text{ at } t) \times a_{ij}^{(h)}
\end{equation}

where $a_{ij}^{(h)}$ is the $(i,j)$ element of $\mathbf{A}^h$.

\textbf{Output:}
\begin{itemize}
    \item Future regime probability distribution $\mathbf{p}_{t+h}$ for each horizon $h$
    \item Evolution of regime probabilities over time
\end{itemize}

\section{Module 2: Economic Variable Forecasting}

\subsection{Component 2.1: Regime-Specific Economic Patterns}

\textbf{Input:}
\begin{itemize}
    \item Historical economic data: $\{y_t\}$ where $y_t$ represents GDP growth, inflation, unemployment, etc.
    \item Historical regime assignments: $\{s_t\}$ where $s_t \in \{0,1,2,3\}$
\end{itemize}

\textbf{Process:}
For each economic variable and each regime, compute regime-specific statistics:
\begin{align}
    \mu_{y,j} &= \mathbb{E}[y_t | s_t = j] \\
    \sigma_{y,j}^2 &= \text{Var}[y_t | s_t = j]
\end{align}

Assuming normality:
\begin{equation}
    y_t | s_t = j \sim \mathcal{N}(\mu_{y,j}, \sigma_{y,j}^2)
\end{equation}

\textbf{Output:}
\begin{itemize}
    \item Regime-specific distributions for each economic variable
    \item Mean and variance parameters: $\{(\mu_{y,j}, \sigma_{y,j}^2)\}_{j=0}^{3}$
\end{itemize}

\subsection{Component 2.2: Variable Importance Matrix}

\textbf{Input:}
\begin{itemize}
    \item Out-of-sample R² results: $R^2_{i,j,h}$ for variable $i$, regime $j$, horizon $h$
    \item Direction accuracy: $\text{DA}_{i,j,h}$ for variable $i$, regime $j$, horizon $h$
    \item Statistical significance: $p_{i,j,h}$ (p-value)
\end{itemize}

\textbf{Process:}
Convert empirical results to importance scores:
\begin{equation}
    \text{Importance}_{i,j,h} = \frac{R^2_{i,j,h} - R^2_{\min}}{R^2_{\max} - R^2_{\min}} \times \text{DA}_{i,j,h} \times \mathbf{1}(p_{i,j,h} < 0.05)
\end{equation}

where $R^2_{\min}$ and $R^2_{\max}$ are the minimum and maximum R² values across all variables, and $\mathbf{1}(\cdot)$ is an indicator function for statistical significance.

\textbf{Output:}
\begin{itemize}
    \item Variable importance matrix: $\mathbf{I} \in \mathbb{R}^{N \times 4 \times 4}$
    \item Where $N$ is the number of macro variables, and dimensions are (Variable, Regime, Horizon)
\end{itemize}

\subsection{Component 2.3: Expected Significance Calculation}

\textbf{Input:}
\begin{itemize}
    \item Future regime probabilities $\mathbf{p}_{t+h}$ (from Module 1, Component 1.3)
    \item Variable importance matrix $\mathbf{I}$ (from Component 2.2)
\end{itemize}

\textbf{Process:}
For each variable $i$ and horizon $h$, compute expected significance:
\begin{equation}
    \text{ES}_{i,h} = \sum_{j=0}^{3} P(\text{Regime}_j \text{ at } t+h) \times \text{Importance}_{i,j,h}
\end{equation}

This weights each variable's importance by the probability of being in regimes where it matters most.

\textbf{Output:}
\begin{itemize}
    \item Expected significance for each variable at each horizon: $\{\text{ES}_{i,h}\}$
    \item Ranked list of variables by expected significance
\end{itemize}

\subsection{Component 2.4: Regime-Based Economic Forecast}

\textbf{Input:}
\begin{itemize}
    \item Future regime probabilities $\mathbf{p}_{t+h}$ (from Module 1, Component 1.3)
    \item Regime-specific economic patterns $\{(\mu_{y,j}, \sigma_{y,j}^2)\}$ (from Component 2.1)
\end{itemize}

\textbf{Process:}
For each economic variable $y$ and horizon $h$, compute the expected value:
\begin{equation}
    \hat{y}_{t+h} = \sum_{j=0}^{3} P(\text{Regime}_j \text{ at } t+h) \times \mu_{y,j}
\end{equation}

The forecast variance is:
\begin{equation}
    \text{Var}[\hat{y}_{t+h}] = \sum_{j=0}^{3} P(\text{Regime}_j \text{ at } t+h) \times \left[(\mu_{y,j} - \hat{y}_{t+h})^2 + \sigma_{y,j}^2\right]
\end{equation}

\textbf{Output:}
\begin{itemize}
    \item Point forecasts: $\hat{y}_{t+h}$ for each economic variable at each horizon
    \item Forecast distributions with confidence intervals
\end{itemize}

\subsection{Component 2.5: Macro Variable Adjustment}

\textbf{Input:}
\begin{itemize}
    \item Regime-based forecast $\hat{y}_{t+h}$ (from Component 2.4)
    \item Current macro variable values: $\{x_{i,t}\}$
    \item Expected significance weights $\{\text{ES}_{i,h}\}$ (from Component 2.3)
    \item Historical means: $\{\bar{x}_i\}$
    \item Sensitivity coefficients: $\{\beta_{i,y}\}$ (from regression analysis)
\end{itemize}

\textbf{Process:}
Compute adjustment term:
\begin{equation}
    \text{Adjustment}_{t+h} = \sum_{i=1}^{N} \text{ES}_{i,h} \times (x_{i,t} - \bar{x}_i) \times \beta_{i,y}
\end{equation}

Final forecast:
\begin{equation}
    \tilde{y}_{t+h} = \hat{y}_{t+h} + \text{Adjustment}_{t+h}
\end{equation}

\textbf{Output:}
\begin{itemize}
    \item Adjusted economic forecasts: $\tilde{y}_{t+h}$
    \item Contribution of each macro variable to the forecast
\end{itemize}

\subsection{Component 2.6: Forecast Accuracy Evaluation}

\textbf{Input:}
\begin{itemize}
    \item Historical forecasts: $\{\tilde{y}_{t+h}\}$
    \item Actual realized values: $\{y_{t+h}\}$
\end{itemize}

\textbf{Process:}
Compute forecast accuracy metrics:

\begin{align}
    \text{RMSE} &= \sqrt{\frac{1}{T} \sum_{t=1}^{T} (y_{t+h} - \tilde{y}_{t+h})^2} \\
    \text{MAE} &= \frac{1}{T} \sum_{t=1}^{T} |y_{t+h} - \tilde{y}_{t+h}| \\
    \text{Direction Accuracy} &= \frac{1}{T} \sum_{t=1}^{T} \mathbf{1}(\text{sign}(y_{t+h}) = \text{sign}(\tilde{y}_{t+h})) \\
    R^2_{\text{OOS}} &= 1 - \frac{\sum_{t=1}^{T} (y_{t+h} - \tilde{y}_{t+h})^2}{\sum_{t=1}^{T} (y_{t+h} - \bar{y})^2}
\end{align}

\textbf{Benchmarks:}
\begin{itemize}
    \item Historical average: $\hat{y}_{t+h} = \bar{y}$
    \item Random walk: $\hat{y}_{t+h} = y_t$
    \item Unweighted model: Equal weights for all variables
\end{itemize}

\textbf{Output:}
\begin{itemize}
    \item Forecast accuracy metrics
    \item Comparison to benchmarks
    \item Component contribution analysis
\end{itemize}

\section{Module 3: Asset Allocation Engine}

\subsection{Component 3.1: Economic Forecast Interpretation}

\textbf{Input:}
\begin{itemize}
    \item Economic forecasts $\tilde{y}_{t+h}$ (from Module 2, Component 2.5)
    \item Forecast horizons $h \in \{1, 3, 6, 12\}$ months
\end{itemize}

\textbf{Process:}
Categorize forecasts into allocation signals:

\begin{align}
    \text{Signal}_{\text{GDP}} &= \begin{cases}
        \text{Risk-on} & \text{if } \tilde{y}_{\text{GDP},t+h} > 2.5\% \\
        \text{Neutral} & \text{if } 1.5\% \leq \tilde{y}_{\text{GDP},t+h} \leq 2.5\% \\
        \text{Risk-off} & \text{if } \tilde{y}_{\text{GDP},t+h} < 1.5\%
    \end{cases} \\
    \text{Signal}_{\text{Inflation}} &= \begin{cases}
        \text{Inflation hedge} & \text{if } \tilde{y}_{\text{Inflation},t+h} > 3\% \\
        \text{Neutral} & \text{if } 1.5\% \leq \tilde{y}_{\text{Inflation},t+h} \leq 3\% \\
        \text{Deflation risk} & \text{if } \tilde{y}_{\text{Inflation},t+h} < 1.5\%
    \end{cases} \\
    \text{Signal}_{\text{Unemployment}} &= \begin{cases}
        \text{Defensive} & \text{if } \Delta \tilde{y}_{\text{Unemployment},t+h} > 0 \\
        \text{Neutral} & \text{if } \Delta \tilde{y}_{\text{Unemployment},t+h} = 0 \\
        \text{Risk-on} & \text{if } \Delta \tilde{y}_{\text{Unemployment},t+h} < 0
    \end{cases}
\end{align}

\textbf{Output:}
\begin{itemize}
    \item Allocation signals from economic forecasts
    \item Signal strength (weak/medium/strong)
\end{itemize}

\subsection{Component 3.2: Return Forecast Interpretation}

\textbf{Input:}
\begin{itemize}
    \item Expected significance weights $\{\text{ES}_{i,h}\}$ (from Module 2, Component 2.3)
    \item Current macro variable values: $\{x_{i,t}\}$
    \item Regression coefficients: $\{\beta_i\}$ from ERP prediction model
\end{itemize}

\textbf{Process:}
Forecast ERP using weighted macro variables:
\begin{equation}
    \widehat{\text{ERP}}_{t+h} = \alpha + \sum_{i=1}^{N} \text{ES}_{i,h} \times \beta_i \times x_{i,t}
\end{equation}

Interpret for allocation:
\begin{equation}
    \text{Signal}_{\text{ERP}} = \begin{cases}
        \text{Increase equity} & \text{if } \widehat{\text{ERP}}_{t+h} > 5\% \\
        \text{Neutral equity} & \text{if } 3\% \leq \widehat{\text{ERP}}_{t+h} \leq 5\% \\
        \text{Reduce equity} & \text{if } \widehat{\text{ERP}}_{t+h} < 3\%
    \end{cases}
\end{equation}

\textbf{Output:}
\begin{itemize}
    \item ERP forecast: $\widehat{\text{ERP}}_{t+h}$
    \item Equity allocation signal
\end{itemize}

\subsection{Component 3.3: Regime-Based Risk Management}

\textbf{Input:}
\begin{itemize}
    \item Current and future regime probabilities $\mathbf{p}_t, \mathbf{p}_{t+h}$ (from Module 1)
    \item Historical portfolio performance by regime
\end{itemize}

\textbf{Process:}
Risk adjustment based on regime probabilities:
\begin{equation}
    \text{Risk Adjustment} = \begin{cases}
        -10\% \text{ equity} & \text{if } P(\text{Regime}_0 \text{ at } t+h) > 0.5 \\
        +5\% \text{ equity} & \text{if } P(\text{Regime}_2 \text{ at } t+h) > 0.5 \\
        0\% & \text{otherwise}
    \end{cases}
\end{equation}

Regime transition risk:
\begin{equation}
    \text{Transition Risk} = -5\% \times P(\text{Regime}_0 \text{ at } t+h | \text{Regime}_i \text{ at } t)
\end{equation}

\textbf{Output:}
\begin{itemize}
    \item Risk management signals
    \item Recommended defensive adjustments
\end{itemize}

\subsection{Component 3.4: Allocation Rule Engine}

\textbf{Input:}
\begin{itemize}
    \item Economic forecast signals (from Component 3.1)
    \item Return forecast signals (from Component 3.2)
    \item Risk management signals (from Component 3.3)
    \item Current portfolio allocation: $\mathbf{w}_t = [w_{\text{Equity}}, w_{\text{Bonds}}, w_{\text{Cash}}]$
\end{itemize}

\textbf{Process:}
Combine signals into allocation adjustments:
\begin{equation}
    \Delta w_{\text{Equity}} = \Delta w_{\text{GDP}} + \Delta w_{\text{ERP}} + \Delta w_{\text{Risk}}
\end{equation}

where:
\begin{align}
    \Delta w_{\text{GDP}} &= \begin{cases}
        -5\% & \text{if } \text{Signal}_{\text{GDP}} = \text{Risk-off} \\
        +5\% & \text{if } \text{Signal}_{\text{GDP}} = \text{Risk-on} \\
        0\% & \text{otherwise}
    \end{cases} \\
    \Delta w_{\text{ERP}} &= \begin{cases}
        -10\% & \text{if } \widehat{\text{ERP}}_{t+h} < 3\% \\
        +5\% & \text{if } \widehat{\text{ERP}}_{t+h} > 5\% \\
        0\% & \text{otherwise}
    \end{cases} \\
    \Delta w_{\text{Risk}} &= \text{Risk Adjustment} + \text{Transition Risk}
\end{align}

Final allocation:
\begin{equation}
    \mathbf{w}_{t+h} = \mathbf{w}_t + \Delta \mathbf{w}
\end{equation}

with constraints: $\sum w_i = 1$ and $w_i \geq 0$ for all $i$.

\textbf{Output:}
\begin{itemize}
    \item Target allocation: $\mathbf{w}_{t+h}$
    \item Rebalancing recommendations
\end{itemize}

\subsection{Component 3.5: Sector Allocation}

\textbf{Input:}
\begin{itemize}
    \item Economic forecasts $\tilde{y}_{t+h}$ (from Module 2)
    \item Sector sensitivity to economic variables: $\{\gamma_{s,y}\}$ where $s$ indexes sectors
\end{itemize}

\textbf{Process:}
Map economic forecasts to sector preferences:
\begin{equation}
    w_{s,t+h} = w_{s,\text{benchmark}} + \sum_{y} \gamma_{s,y} \times (\tilde{y}_{y,t+h} - \bar{y}_y)
\end{equation}

where $w_{s,\text{benchmark}}$ is the benchmark sector weight.

Example sector sensitivities:
\begin{align}
    \gamma_{\text{Industrials},\text{GDP}} &= +0.05 \\
    \gamma_{\text{Utilities},\text{GDP}} &= -0.03 \\
    \gamma_{\text{Energy},\text{Inflation}} &= +0.02
\end{align}

\textbf{Output:}
\begin{itemize}
    \item Sector allocation weights: $\{w_{s,t+h}\}$
    \item Sector tilts relative to benchmark
\end{itemize}

\subsection{Component 3.6: Portfolio Backtesting}

\textbf{Input:}
\begin{itemize}
    \item Historical allocation decisions $\{\mathbf{w}_t\}$ (from Components 3.4, 3.5)
    \item Historical asset returns: $\{\mathbf{r}_t\}$
    \item Transaction costs: $c$ (e.g., 0.1\% per trade)
\end{itemize}

\textbf{Process:}
Simulate portfolio performance:
\begin{equation}
    R_{p,t} = \sum_{i} w_{i,t} \times r_{i,t} - c \times \sum_{i} |w_{i,t} - w_{i,t-1}|
\end{equation}

Portfolio value:
\begin{equation}
    V_t = V_{t-1} \times (1 + R_{p,t})
\end{equation}

Performance metrics:
\begin{align}
    \text{Annualized Return} &= \left(\frac{V_T}{V_0}\right)^{1/T} - 1 \\
    \text{Sharpe Ratio} &= \frac{\bar{R}_p - r_f}{\sigma_{R_p}} \\
    \text{Maximum Drawdown} &= \max_t \left(\frac{V_{\max} - V_t}{V_{\max}}\right)
\end{align}

\textbf{Benchmark:}
\begin{itemize}
    \item Buy-and-hold 60/40 portfolio: $w_{\text{Equity}} = 0.6$, $w_{\text{Bonds}} = 0.4$
    \item Static allocation
\end{itemize}

\textbf{Output:}
\begin{itemize}
    \item Portfolio performance metrics
    \item Comparison to benchmark
    \item Performance attribution analysis
\end{itemize}

\section{Model Integration Flow}

The complete model flow is:

\begin{align}
    \text{Module 1} &\rightarrow \mathbf{p}_{t+h} \text{ (Future regime probabilities)} \\
    \mathbf{p}_{t+h} + \text{Historical patterns} &\rightarrow \hat{y}_{t+h} \text{ (Regime-based forecasts)} \\
    \mathbf{p}_{t+h} + \text{Importance matrix} &\rightarrow \text{ES}_{i,h} \text{ (Expected significance)} \\
    \hat{y}_{t+h} + \text{ES}_{i,h} + \text{Macro variables} &\rightarrow \tilde{y}_{t+h} \text{ (Adjusted forecasts)} \\
    \tilde{y}_{t+h} + \text{ERP forecast} + \mathbf{p}_{t+h} &\rightarrow \text{Allocation signals} \\
    \text{Allocation signals} &\rightarrow \mathbf{w}_{t+h} \text{ (Portfolio weights)} \\
    \mathbf{w}_{t+h} + \text{Returns} &\rightarrow \text{Portfolio performance}
\end{align}
